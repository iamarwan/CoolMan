def hash_password(password: str) -> str:
    salt = secrets.token_hex(16)
    dk = hashlib.pbkdf2_hmac("sha256", password.encode(), salt.encode(), _PBKDF2_ITERATIONS)
    return f"{salt}${dk.hex()}"


def verify_password(password: str, hashed: str) -> bool:
    try:
        salt, dk_hex = hashed.split("$", 1)
        dk = hashlib.pbkdf2_hmac("sha256", password.encode(), salt.encode(), _PBKDF2_ITERATIONS)
        return secrets.compare_digest(dk.hex(), dk_hex)
    except Exception:
        return False


# ─────────────────────────────────────────────
# JWT PRIMITIVES
# ─────────────────────────────────────────────

def _b64url_encode(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")


def _b64url_decode(data: str) -> bytes:
    padded = data + "=" * (-len(data) % 4)
    return base64.urlsafe_b64decode(padded.encode("ascii"))


def _sign(signing_input: str) -> str:
    digest = hmac.new(
        JWT_SECRET.encode("utf-8"),
        signing_input.encode("ascii"),
        hashlib.sha256,
    ).digest()
    return _b64url_encode(digest)


def _build_token(sub: str, role: str, audience: str, expires_minutes: int) -> str:
    now = datetime.now(timezone.utc)
    exp = now + timedelta(minutes=expires_minutes)
    header = {"alg": "HS256", "typ": "JWT"}
    payload = {
        "sub": sub,
        "role": role,
        "iss": JWT_ISSUER,
        "aud": audience,
        "iat": int(now.timestamp()),
        "exp": int(exp.timestamp()),
    }
    encoded_header = _b64url_encode(
        json.dumps(header, separators=(",", ":"), sort_keys=True).encode("utf-8")
    )
    encoded_payload = _b64url_encode(
        json.dumps(payload, separators=(",", ":"), sort_keys=True).encode("utf-8")
    )
    signing_input = f"{encoded_header}.{encoded_payload}"
    return f"{signing_input}.{_sign(signing_input)}"


def _decode_and_validate(token: str, expected_audience: str, expected_role: str) -> dict:
    try:
        header_part, payload_part, signature_part = token.split(".")
    except ValueError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token format.",
        ) from exc

    signing_input = f"{header_part}.{payload_part}"
    if not secrets.compare_digest(_sign(signing_input), signature_part):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token signature.",
        )

    try:
        payload = json.loads(_b64url_decode(payload_part))
    except Exception as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token payload.",
        ) from exc

    now_ts = int(datetime.now(timezone.utc).timestamp())
    if payload.get("exp", 0) < now_ts:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Token expired.")
    if payload.get("iss") != JWT_ISSUER or payload.get("aud") != expected_audience:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid token claims.")
    if payload.get("role") != expected_role:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail=f"{expected_role.capitalize()} token required.",
        )
    return payload

## 🏗️ 1. Architecture Overview

The system relies on a two-tier full-stack architecture:
1.  **Mobile Client (Flutter/Dart)**: An immersive single-page map controller with visual overlays, legend overlays, bottom detail drawers, and onboarding pages. Recharts OSM map tiles with high-performance marker clustering (`flutter_map_marker_cluster`) to avoid slow layouts.
2.  **Server Engine (FastAPI/SQLAlchemy)**: A fully asynchronous backend processing REST queries, rate limiter, and authentication. Coordinates GIS spatial filters via PostGIS native SQL directives.

---

## 🗄️ 2. Database Schema & Relations

The PostgreSQL schema is structured around 4 interactive entities:

```
                  ┌──────────────┐
                  │    users     │
                  └──────┬───────┘
                         │ 1
                         │
                         │ 1..* (Submissions, Approvals, Votes)
                         ▼
  ┌──────────┐ 1      ┌─────────────────┐ 1      ┌────────────────┐
  │ sponsors ├───────>│  water_coolers  ├────────>│ cooler_photos  │
  └──────────┘  0..*  └─────────────────┘   0..*  └────────────────┘
```

### Table Definitions

#### `users`
Tracks registered community contributors and administrators.
*   `id` (UUID): Primary Key.
*   `username` (String(64)): Unique index.
*   `hashed_password` (String(256)): Structured password hash.
*   `trust_score` (Integer): Dynamic score mapping permissions. Default: 0.
*   `created_at` (DateTime): Registration date.

#### `sponsors`
Tracks commercial or religious organizations sponsoring charity coolers.
*   `id` (UUID): Primary Key.
*   `name` (String(255)): Entity name.
*   `sponsor_type` (Enum): Business, Mosque, Individual, NGO.
*   `contact` (String(255), optional): Launch links or telephone numbers.
*   `verified` (Boolean): Administrative endorsement state. Default: False.

#### `water_coolers`
The main spatial entity representing physical drinking water stations.
*   `id` (UUID): Primary Key.
*   `name` (String(255)): Human-readable label or landmark description.
*   `latitude`, `longitude` (Numeric(10, 7)): Decimal map coordinates.
*   `geom` (Geometry(POINT, 4326)): Spatial coordinate mapped to PostGIS indexes.
*   `status` (Enum): Working, Empty, Broken.
*   `last_verified_at` (DateTime): Last confirmation date.
*   `is_approved` (Boolean): Live visibility state.
*   `is_rejected` (Boolean): Deemed spam or non-existent.
*   `pending_status` (String, optional): Staged update proposed by non-trusted reporter.
*   `has_cups` (Boolean, optional): High-level cup count stock.
*   `cups_type` (Enum, optional): Communal, Disposable.
*   `user_id` (UUID, Foreign Key): Identifier of the original reporter.
*   `pending_status_user_id` (UUID, Foreign Key): Submitter of review updates.
*   `sponsor_id` (UUID, Foreign Key): Linked charity sponsor.

#### `cooler_photos`
Tracks verification photos uploaded for security review.
*   `id` (UUID): Primary Key.
*   `cooler_id` (UUID, Foreign Key): Owning cooler.
*   `file_path` (String(512)): File address on the disk or S3 bucket.
*   `uploaded_at` (DateTime): Time of upload.
*   `is_approved` (Boolean): Admin approved for public view.
*   `photo_category` (Enum): Cooler, Cups.

---

## 🔒 3. Integrity, Authentication & Security Guidelines

### Password Encryption
All client credentials utilize cryptographic standards:
*   Passes are salted and scrambled via **PBKDF2-HMAC-SHA256**.
*   Work factor iterations are set to **260,000 iterations**, avoiding brute-force vectors.

### JWT Structure
Tokens are signed server-side using **HMAC-SHA256**:
*   *Audience parameters*: `sabil-user` or `sabil-admin`.
*   *Expirations*: Standard user token is set to **7 days** (10,080 minutes), while admin tokens auto-invalidate after **8 hours** to limit hijacking windows.

---

## 🎮 4. Verification Mechanics & Trust Logic

A central gamified trust loop handles review queues and defends the database against spam:

1.  **Submission Verification**:
    *   Proposed coolers get queued as `is_approved = False`.
    *   If the submitter is a **Verified Reporter (Trust Score ≥ 500)**, the cooler is auto-approved and goes live instantly.
2.  **Trust Flow Calculations**:
    *   *Add New Cooler*: Approved: **+50 Trust** to the reporter. Rejected: **-100 Trust**.
    *   *Verify / Confirm*: Tap "Still Working" (restricted by 1-hour cooldown per cooler): **+5 Trust**.
    *   *Propose Status Update*: Suggested status change: **+10 Trust** upon submission. If trust ≥ 500, the status updates instantly. Otherwise, it is staged as `pending_status` for admin approval.
3.  **GPS Duplication Guard**:
    *   The database rejects adding a cooler if another cooler exists within a **15-meter radius** (determined via PostGIS `ST_DWithin` on the coordinates index). This blocks multiple reports for the same cooler.

def _cooler_by_id_stmt(cooler_id: str):
    return select(WaterCooler).where(cast(WaterCooler.id, String) == cooler_id)


async def _get_cooler_or_404(cooler_id: str, db: AsyncSession) -> WaterCooler:
    result = await db.execute(_cooler_by_id_stmt(cooler_id))
    cooler = result.scalars().first()
    if not cooler:
        raise HTTPException(status_code=404, detail="Cooler not found")
    return cooler


async def _get_user_or_404(user_id, db: AsyncSession) -> User:
    result = await db.execute(select(User).where(User.id == user_id))
    user = result.scalars().first()
    if not user:
        raise HTTPException(status_code=404, detail="User not found")
    return user

@app.post("/auth/register", response_model=UserAuthResponse, status_code=201)
async def register(payload: UserCreate, db: AsyncSession = Depends(get_db)):
    existing = await db.execute(select(User).where(User.username == payload.username))
    if existing.scalars().first():
        raise HTTPException(status_code=409, detail="Username already taken.")
    user = User(
        username=payload.username,
        hashed_password=hash_password(payload.password),
        trust_score=0,
    )
    db.add(user)
    await db.commit()
    await db.refresh(user)
    token = create_user_token(user.id)
    return {
        "id": str(user.id),
        "username": user.username,
        "trust_score": user.trust_score,
        "access_token": token,
    }


@app.post("/auth/login", response_model=UserAuthResponse)
async def user_login(payload: UserCreate, db: AsyncSession = Depends(get_db)):
    result = await db.execute(select(User).where(User.username == payload.username))
    user = result.scalars().first()
    if not user or not verify_password(payload.password, user.hashed_password):
        raise HTTPException(status_code=401, detail="Invalid credentials.")
    return {
        "access_token": create_user_token(user.id),
        "id": str(user.id),
        "username": user.username,
        "trust_score": user.trust_score,
    }

@app.get("/coolers/my", response_model=list[WaterCoolerRead])
async def get_my_coolers(
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    stmt = (
        select(WaterCooler)
        .where(
            or_(
                WaterCooler.user_id == current_user.id,
                WaterCooler.pending_status_user_id == current_user.id,
            )
        )
        .order_by(WaterCooler.last_verified_at.desc())
    )
    result = await db.execute(stmt)
    coolers = result.scalars().all()
    # Deduplicate: a cooler might match both conditions
    seen: set = set()
    unique = []
    for c in coolers:
        if c.id not in seen:
            seen.add(c.id)
            unique.append(c)
    return unique

@app.get("/coolers/nearby", response_model=list[WaterCoolerRead])
async def get_nearby_coolers(
    lat: float = Query(..., ge=-90, le=90),
    lng: float = Query(..., ge=-180, le=180),
    radius_km: float = Query(5.0, gt=0, le=50),
    db: AsyncSession = Depends(get_db),
):
    target = ST_SetSRID(ST_MakePoint(lng, lat), 4326)
    radius_m = radius_km * 1000

    stmt = select(WaterCooler).where(
        WaterCooler.is_approved == True,
        ST_DWithin(
            cast(WaterCooler.geom, Geography(geometry_type="POINT", srid=4326)),
            cast(target, Geography(geometry_type="POINT", srid=4326)),
            radius_m,
            use_spheroid=True,
        ),
    )
    result = await db.execute(stmt)
    return result.scalars().all()

@app.patch("/coolers/{cooler_id}/status", response_model=WaterCoolerRead)
async def update_cooler_status(
    cooler_id: str,
    payload: StatusUpdate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    valid_statuses = {"Working", "Empty", "Broken"}
    if payload.status not in valid_statuses:
        raise HTTPException(status_code=400, detail=f"Invalid status. Must be one of: {valid_statuses}")

    cooler = await _get_cooler_or_404(cooler_id, db)

    if current_user.trust_score >= _TRUST_AUTO_APPROVE_THRESHOLD:
        cooler.status = payload.status
        cooler.pending_status = None
        cooler.pending_status_user_id = None
        cooler.last_verified_at = datetime.now(timezone.utc)
        keys_to_delete = [k for k in _confirm_store if k[1] == cooler_id]
        for k in keys_to_delete:
            del _confirm_store[k]
    else:
        cooler.pending_status = payload.status
        cooler.pending_status_user_id = current_user.id

    current_user.trust_score = current_user.trust_score + _STATUS_UPDATE_REWARD

    await db.commit()
    await db.refresh(cooler)
    return cooler


@app.post("/coolers/{cooler_id}/confirm")
async def confirm_cooler_working(
    cooler_id: str,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    key = (str(current_user.id), cooler_id)
    now = time.time()
    last = _confirm_store.get(key, 0.0)

    if now - last < _CONFIRM_COOLDOWN:
        remaining_minutes = int((_CONFIRM_COOLDOWN - (now - last)) / 60)
        raise HTTPException(
            status_code=429,
            detail=f"You can confirm this cooler again in {remaining_minutes} minute{'s' if remaining_minutes != 1 else ''}.",
        )

    cooler = await _get_cooler_or_404(cooler_id, db)
    if not cooler.is_approved:
        raise HTTPException(status_code=400, detail="Cooler is not yet approved.")

    cooler.last_verified_at = datetime.now(timezone.utc)
    current_user.trust_score = current_user.trust_score + _CONFIRM_REWARD
    _confirm_store[key] = now

    await db.commit()
    await db.refresh(current_user)
    return {"message": "Confirmed", "trust_score": current_user.trust_score}


@app.post(
    "/coolers",
    response_model=WaterCoolerRead,
    status_code=201,
    dependencies=[Depends(rate_limiter)],
)
async def create_cooler(
    payload: WaterCoolerCreate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    target = ST_SetSRID(ST_MakePoint(payload.longitude, payload.latitude), 4326)

    dup_result = await db.execute(
        select(WaterCooler).where(
            ST_DWithin(
                cast(WaterCooler.geom, Geography(geometry_type="POINT", srid=4326)),
                cast(target, Geography(geometry_type="POINT", srid=4326)),
                15.0,
                use_spheroid=True,
            )
        )
    )
    if dup_result.scalars().first() is not None:
        raise HTTPException(
            status_code=400,
            detail="A water cooler has already been reported at this exact location.",
        )

    is_auto_approved = current_user.trust_score >= _TRUST_AUTO_APPROVE_THRESHOLD

    cooler = WaterCooler(
        name=payload.name,
        latitude=payload.latitude,
        longitude=payload.longitude,
        geom=WKTElement(f"POINT({payload.longitude} {payload.latitude})", srid=4326),
        status=payload.status,
        is_approved=is_auto_approved,
        has_cups=payload.has_cups,
        cups_type=payload.cups_type,
        user_id=current_user.id,
    )
    db.add(cooler)

    if is_auto_approved:
        current_user.trust_score = current_user.trust_score + _TRUST_APPROVE_REWARD

    await db.commit()
    await db.refresh(cooler)
    return cooler


@app.post(
    "/coolers/{cooler_id}/photos",
    response_model=CoolerPhotoRead,
    status_code=201,
    dependencies=[Depends(rate_limiter)],
)
async def upload_cooler_photo(
    cooler_id: str,
    file: UploadFile = File(...),
    photo_category: str = Form("Cooler"),
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    from models import PhotoCategory as PC

    cooler = await _get_cooler_or_404(cooler_id, db)

    allowed = {"image/jpeg", "image/png", "image/webp"}
    if file.content_type not in allowed:
        raise HTTPException(status_code=400, detail="Invalid file type. Use JPEG, PNG, or WebP.")

    try:
        category = PC(photo_category)
    except ValueError:
        category = PC.Cooler

    ext = file.filename.rsplit(".", 1)[-1] if "." in file.filename else "jpg"
    filename = f"{cooler_id}_{datetime.now(timezone.utc).timestamp():.0f}.{ext}"
    dest = os.path.join(UPLOAD_DIR, filename)

    with open(dest, "wb") as buf:
        shutil.copyfileobj(file.file, buf)

    photo = CoolerPhoto(
        cooler_id=cooler.id,
        file_path=f"/uploads/photos/{filename}",
        photo_category=category,
        user_id=current_user.id,
    )
    db.add(photo)
    await db.commit()
    await db.refresh(photo)
    return photo

@app.patch("/admin/coolers/{cooler_id}/approve", dependencies=[Depends(require_admin)])
async def approve_cooler(cooler_id: str, db: AsyncSession = Depends(get_db)):
    cooler = await _get_cooler_or_404(cooler_id, db)

    if cooler.pending_status:
        cooler.status = cooler.pending_status
        cooler.pending_status = None
        cooler.pending_status_user_id = None
        keys_to_delete = [k for k in _confirm_store if k[1] == cooler_id]
        for k in keys_to_delete:
            del _confirm_store[k]

    if not cooler.is_approved and not cooler.is_rejected:
        cooler.is_approved = True
        for photo in cooler.photos:
            photo.is_approved = True
        if cooler.user_id:
            submitter = await _get_user_or_404(cooler.user_id, db)
            submitter.trust_score = submitter.trust_score + _TRUST_APPROVE_REWARD

    await db.commit()
    return {"message": "Approved"}


@app.patch("/admin/coolers/{cooler_id}/reject", dependencies=[Depends(require_admin)])
async def reject_cooler(cooler_id: str, db: AsyncSession = Depends(get_db)):
    cooler = await _get_cooler_or_404(cooler_id, db)

    if not cooler.is_approved and not cooler.is_rejected and cooler.user_id:
        submitter = await _get_user_or_404(cooler.user_id, db)
        submitter.trust_score = max(0, submitter.trust_score + _TRUST_REJECT_PENALTY)

    cooler.is_rejected = True
    cooler.is_approved = False
    cooler.pending_status = None
    cooler.pending_status_user_id = None
    await db.commit()
    return {"message": "Rejected"}

@app.get("/sponsors", response_model=list[SponsorReadWithCount])
async def list_sponsors(db: AsyncSession = Depends(get_db)):
    count_subq = (
        select(
            WaterCooler.sponsor_id,
            func.count(WaterCooler.id).label("cooler_count"),
        )
        .where(WaterCooler.is_approved == True, WaterCooler.sponsor_id.is_not(None))
        .group_by(WaterCooler.sponsor_id)
        .subquery()
    )

    stmt = (
        select(Sponsor, func.coalesce(count_subq.c.cooler_count, 0).label("cooler_count"))
        .outerjoin(count_subq, Sponsor.id == count_subq.c.sponsor_id)
        .order_by(Sponsor.name)
    )
    rows = (await db.execute(stmt)).all()

    return [
        SponsorReadWithCount(
            id=sponsor.id,
            name=sponsor.name,
            sponsor_type=sponsor.sponsor_type,
            contact=sponsor.contact,
            verified=sponsor.verified,
            created_at=sponsor.created_at,
            cooler_count=count,
        )
        for sponsor, count in rows
    ]


@app.get("/sponsors/{sponsor_id}", response_model=SponsorReadWithCount)
async def get_sponsor(sponsor_id: str, db: AsyncSession = Depends(get_db)):
    result = await db.execute(
        select(Sponsor).where(cast(Sponsor.id, String) == sponsor_id)
    )
    sponsor = result.scalars().first()
    if not sponsor:
        raise HTTPException(status_code=404, detail="Sponsor not found")

    count_result = await db.execute(
        select(func.count(WaterCooler.id)).where(
            WaterCooler.sponsor_id == sponsor.id,
            WaterCooler.is_approved == True,
        )
    )
    count = count_result.scalar() or 0
    return SponsorReadWithCount(
        id=sponsor.id,
        name=sponsor.name,
        sponsor_type=sponsor.sponsor_type,
        contact=sponsor.contact,
        verified=sponsor.verified,
        created_at=sponsor.created_at,
        cooler_count=count,
    )

import 'dart:async';
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:image_picker/image_picker.dart';
import '../models/water_cooler.dart';
import '../services/api_service.dart';

class AdminScreen extends StatefulWidget {
  const AdminScreen({super.key});

  @override
  State<AdminScreen> createState() => _AdminScreenState();
}

class _AdminScreenState extends State<AdminScreen>
    with SingleTickerProviderStateMixin {
  final ApiService _api = ApiService();
  final ImagePicker _imagePicker = ImagePicker();

  late final TabController _tabController;
  final TextEditingController _searchController = TextEditingController();
  bool _authorized = false;

  List<WaterCooler> _pending = [];
  List<WaterCooler> _current = [];
  List<Sponsor> _sponsors = [];
  
  bool _loadingPending = true;
  bool _loadingCurrent = true;
  bool _loadingSponsors = true;
  
  String _query = '';
  Timer? _debounce;

  @override
  void initState() {
    super.initState();
    _tabController = TabController(length: 3, vsync: this);
    _searchController.addListener(_onSearchChanged);
    WidgetsBinding.instance.addPostFrameCallback((_) => _authorizeAndLoad());
  }

  void _onSearchChanged() {
    _debounce?.cancel();
    _debounce = Timer(const Duration(milliseconds: 500), () {
      if (mounted) {
        setState(() => _query = _searchController.text.trim().toLowerCase());
      }
    });
  }

  @override
  void dispose() {
    _debounce?.cancel();
    _tabController.dispose();
    _searchController.removeListener(_onSearchChanged);
    _searchController.dispose();
    super.dispose();
  }

  Future<void> _fetchPending() async {
    if (!_authorized) return;
    setState(() => _loadingPending = true);
    try {
      final result = await _api.fetchPendingCoolers();
      setState(() {
        _pending = result;
        _loadingPending = false;
      });
    } catch (e) {
      setState(() => _loadingPending = false);
      if (_handleAdminAuthError(e)) return;
      _snack('Failed to load pending coolers.', isError: true);
    }
  }

  Future<void> _fetchCurrent() async {
    if (!_authorized) return;
    setState(() => _loadingCurrent = true);
    try {
      final result = await _api.fetchCurrentCoolers();
      setState(() {
        _current = result;
        _loadingCurrent = false;
      });
    } catch (e) {
      setState(() => _loadingCurrent = false);
      if (_handleAdminAuthError(e)) return;
      _snack('Failed to load live coolers.', isError: true);
    }
  }

  Future<void> _fetchSponsors() async {
    if (!_authorized) return;
    setState(() => _loadingSponsors = true);
    try {
      final result = await _api.fetchAllSponsors();
      setState(() {
        _sponsors = result;
        _loadingSponsors = false;
      });
    } catch (e) {
      setState(() => _loadingSponsors = false);
      if (_handleAdminAuthError(e)) return;
      _snack('Failed to load sponsors.', isError: true);
    }
  }

  Future<void> _approve(WaterCooler cooler) async {
    setState(() => _pending.removeWhere((c) => c.id == cooler.id));
    try {
      if (cooler.isApproved) {
        // Already approved cooler, so this must be a photo review card.
        // Approve all unapproved photos for this cooler.
        for (final p in cooler.photos) {
          if (!p.isApproved) {
            await _api.approvePhoto(p.id);
          }
        }
        _snack('✓ Photos for "${cooler.name}" approved.');
      } else {
        await _api.approveCooler(cooler.id);
        _snack('✓ "${cooler.name}" approved.');
      }
      _fetchCurrent();
    } catch (e) {
      if (_handleAdminAuthError(e)) return;
      _snack('Approval failed — refreshing.', isError: true);
      _fetchPending();
    }
  }

  Future<void> _reject(WaterCooler cooler) async {
    setState(() => _pending.removeWhere((c) => c.id == cooler.id));
    try {
      if (cooler.pendingStatus != null) {
        await _api.rejectStatusChange(cooler.id);
        _snack('"${cooler.name}" status update rejected.');
      } else if (cooler.isApproved) {
        // Already approved cooler, so this must be a photo review card.
        // Delete all unapproved photos for this cooler.
        for (final p in cooler.photos) {
          if (!p.isApproved) {
            await _api.deletePhoto(p.id);
          }
        }
        _snack('Photos for "${cooler.name}" rejected.');
      } else {
        await _api.rejectCooler(cooler.id);
        _snack('"${cooler.name}" rejected.');
      }
    } catch (e) {
      if (_handleAdminAuthError(e)) return;
      _snack('Rejection failed — refreshing.', isError: true);
      _fetchPending();
    }
  }

  Future<void> _deleteSpam(WaterCooler cooler) async {
    setState(() => _current.removeWhere((c) => c.id == cooler.id));
    try {

    } catch (e) {
      if (_handleAdminAuthError(e)) return;

      _fetchCurrent();
    }
  }

class _PhotoGallerySheetState extends State<_PhotoGallerySheet> {
  static const _base = 'http://10.0.2.2:8000';
  List<CoolerPhoto> _photos = [];
  bool _loading = true;
  String? _error;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    try {
      final photos = await widget.api.fetchCoolerPhotosAdmin(widget.cooler.id);
      setState(() {
        _photos = photos;
        _loading = false;
      });
    } catch (e) {
      setState(() {
        _error = 'Failed to load photos.';
        _loading = false;
      });
    }
  }

  Future<void> _approvePhoto(CoolerPhoto photo) async {
    try {
      await widget.api.approvePhoto(photo.id);
      setState(() {
        final idx = _photos.indexWhere((p) => p.id == photo.id);
        if (idx != -1) {
          _photos[idx] = CoolerPhoto(
            id: photo.id,
            coolerId: photo.coolerId,
            filePath: photo.filePath,
            uploadedAt: photo.uploadedAt,
            isApproved: true,
          );
        }
      });
    } catch (_) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
              content: const Text('Approval failed.'),
              backgroundColor: Colors.red.shade700),
        );
      }
    }
  }

  Future<void> _deletePhoto(CoolerPhoto photo) async {
    try {
      await widget.api.deletePhoto(photo.id);
      setState(() => _photos.removeWhere((p) => p.id == photo.id));
    } catch (_) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
              content: const Text('Delete failed.'),
              backgroundColor: Colors.red.shade700),
        );
      }
    }
  }

class _SponsorAssignSheet extends StatefulWidget {
  final WaterCooler cooler;
  final ApiService api;
  final List<Sponsor> sponsors;
  final VoidCallback onAssigned;
  final VoidCallback onRemoved;

  const _SponsorAssignSheet({
    required this.cooler,
    required this.api,
    required this.sponsors,
    required this.onAssigned,
    required this.onRemoved,
  });

  @override
  State<_SponsorAssignSheet> createState() => _SponsorAssignSheetState();
}

class _SponsorAssignSheetState extends State<_SponsorAssignSheet> {
  bool _loading = false;
  String? _error;

  Future<void> _removeSponsor() async {
    setState(() {
      _loading = true;
      _error = null;
    });

    try {
      await widget.api.removeSponsor(widget.cooler.id);
      widget.onRemoved();
      if (mounted) Navigator.of(context).pop();
    } catch (e) {
      setState(() {
        _loading = false;
        _error = 'Failed to remove sponsor association.';
      });
    }
  }

  Future<void> _assign(String id) async {
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      await widget.api.assignSponsor(coolerId: widget.cooler.id, sponsorId: id);
      widget.onAssigned();
      if (mounted) Navigator.of(context).pop();
    } catch (_) {
      setState(() {
        _loading = false;
        _error = 'Failed to assign sponsor.';
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    final sponsor = widget.cooler.sponsor;

    return Container(
      decoration: const BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.vertical(top: Radius.circular(24)),
      ),
      constraints: BoxConstraints(
        maxHeight: MediaQuery.of(context).size.height * 0.6,
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          const SizedBox(height: 12),
          Center(

class _SwipeablePendingCard extends StatelessWidget {
  final WaterCooler cooler;
  final VoidCallback onApprove;
  final VoidCallback onReject;
  final VoidCallback onViewPhotos;

  const _SwipeablePendingCard({
    super.key,
    required this.cooler,
    required this.onApprove,
    required this.onReject,
    required this.onViewPhotos,
  });

  @override
  Widget build(BuildContext context) {
    return Dismissible(
      key: ValueKey(cooler.id),
      direction: DismissDirection.horizontal,
      confirmDismiss: (direction) async {
        if (direction == DismissDirection.startToEnd) {
          onApprove();
          return true;
        } else {
          onReject();
          return true;
        }
      },
      background: _SwipeBg(
        color: const Color(0xFF10B981),
        icon: Icons.check_circle_rounded,
        label: 'Approve',
        alignment: Alignment.centerLeft,
      ),
      secondaryBackground: _SwipeBg(
        color: const Color(0xFFEF4444),
        icon: Icons.cancel_rounded,
        label: 'Reject',
        alignment: Alignment.centerRight,
      ),
      child: _PendingCard(
        cooler: cooler,
        onApprove: onApprove,
        onReject: onReject,
        onViewPhotos: onViewPhotos,
      ),
    );
  }
}

class _PendingCard extends StatelessWidget {
  final WaterCooler cooler;
  final VoidCallback onApprove;
  final VoidCallback onReject;
  final VoidCallback onViewPhotos;

    required this.cooler,
    required this.onApprove,
    required this.onReject,
    required this.onViewPhotos,
  });

  bool get _isStatusUpdate => cooler.pendingStatus != null;
  bool get _isNewProposal => !cooler.isApproved;

  Color get _typeColor {
    if (_isStatusUpdate) return const Color(0xFFF59E0B);
    if (_isNewProposal) return const Color(0xFF4F46E5);
    return const Color(0xFF0F766E);

  IconData get _typeIcon {
    if (_isStatusUpdate) return Icons.swap_horiz_rounded;
    if (_isNewProposal) return Icons.add_location_alt_rounded;
    return Icons.image_search_rounded;

    if (_isStatusUpdate) return 'Status Update Review';
    if (_isNewProposal) return 'New Location Proposal';

            '${verified.hour.toString().padLeft(2, '0')}:${verified.minute.toString().padLeft(2, '0')}'
        : cooler.lastVerifiedAt;

    return Container(
      margin: const EdgeInsets.only(bottom: 14),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(18),
        boxShadow: [

        crossAxisAlignment: CrossAxisAlignment.start,

              color: _typeColor.withValues(alpha: 0.08),

  Timer? _searchDebounce;

  @override
  void initState() {
    super.initState();
    _searchController.addListener(_onSearchChanged);
    _initLocation();
  }

  void _onSearchChanged() {
    _searchDebounce?.cancel();
    _searchDebounce = Timer(const Duration(milliseconds: 500), () {
      if (mounted && _searchMode) _runSearch(_searchController.text);
    });
  }

  @override
  void dispose() {
    _searchDebounce?.cancel();
    _searchController.removeListener(_onSearchChanged);
    _searchController.dispose();
    super.dispose();
  }

  void _handleTitleTap() {
    final now = DateTime.now();
    if (_firstTapTime == null ||
        now.difference(_firstTapTime!) > const Duration(seconds: 3)) {
      _firstTapTime = now;
      _titleTapCount = 1;
    } else {
      _titleTapCount++;
    }
    if (_titleTapCount >= 5) {
      _titleTapCount = 0;
      _firstTapTime = null;
      Navigator.of(context)
          .push(MaterialPageRoute(builder: (_) => const AdminScreen()));
    }
  }

  Future<void> _initLocation() async {
    final loc = await _resolveLocation();
    setState(() => _userLocation = loc);
    await _fetchCoolers();
  }

  Future<LatLng> _resolveLocation() async {
    try {
      final position = await _location.resolveCurrentPosition();
      if (position != null) return LatLng(position.latitude, position.longitude);
    } on LocationServiceDisabledException catch (e) {
      _showSnackbar(e.message, isError: true);
    } on LocationPermissionDeniedException catch (e) {
      _showSnackbar(e.message, isError: true);
    } on LocationPermissionPermanentlyDeniedException catch (e) {
      _showSnackbar(e.message, isError: true);
    } catch (e) {
      _showSnackbar('Could not get location: $e', isError: true);
    }
    return _fallback;
  }

  Future<void> _fetchCoolers() async {
    final loc = _userLocation ?? _fallback;
    setState(() => _loading = true);
    try {
      final coolers = await _api.fetchNearbyCoolers(
          lat: loc.latitude, lng: loc.longitude);
      setState(() {
        _coolers = coolers;
        _loading = false;
      });
    } catch (e) {
      setState(() => _loading = false);
      _showSnackbar('Failed to load coolers: $e', isError: true);
    }
  }

  Future<void> _recenterOnUser() async {
    final loc = await _resolveLocation();
    setState(() => _userLocation = loc);
    _mapController.move(loc, 14.5);
    await _fetchCoolers();
  }

  Future<void> _runSearch(String query) async {
    if (query.trim().isEmpty) {
      setState(() {
        _searchResults = [];
        _showSearchList = false;
      });
      return;
    }
    setState(() => _searching = true);
    try {
      final results = await _api.searchCoolers(query);
      setState(() {
        _searchResults = results;
        _searching = false;
        _showSearchList = results.isNotEmpty;
      });
      if (results.isNotEmpty) {
        _mapController.move(
            LatLng(results.first.latitude, results.first.longitude), 15.0);
      }
    } catch (e) {
      setState(() => _searching = false);
      _showSnackbar('Search failed: $e', isError: true);
    }
  }

  Future<void> _launchDirections(WaterCooler cooler) async {
    final lat = cooler.latitude;
    final lng = cooler.longitude;
    final androidUrl =
        'https://www.google.com/maps/dir/?api=1&destination=$lat,$lng&travelmode=walking';
    final iosUrl = 'maps://?daddr=$lat,$lng&dirflg=w';
    final fallbackUrl =
        'https://www.openstreetmap.org/directions?from=&to=$lat,$lng';
    final uri = Uri.parse(Platform.isIOS ? iosUrl : androidUrl);
    final fallback = Uri.parse(fallbackUrl);
    try {
      if (await canLaunchUrl(uri)) {
        await launchUrl(uri, mode: LaunchMode.externalApplication);
      } else {
        await launchUrl(fallback, mode: LaunchMode.externalApplication);
      }
    } catch (_) {
      _showSnackbar('Could not open map navigation.', isError: true);
    }
  }

  void _showSnackbar(String message, {bool isError = false}) {

class _AuthDialog extends StatefulWidget {
  const _AuthDialog();
  @override
  State<_AuthDialog> createState() => _AuthDialogState();
}

class _AuthDialogState extends State<_AuthDialog>
    with SingleTickerProviderStateMixin {
  late final TabController _tabs;
  final _loginUserCtrl = TextEditingController();
  final _loginPassCtrl = TextEditingController();
  final _regUserCtrl = TextEditingController();
  final _regPassCtrl = TextEditingController();
  final _loginKey = GlobalKey<FormState>();
  final _regKey = GlobalKey<FormState>();
  final ApiService _api = ApiService();
  bool _loading = false;
  String? _error;

  @override
  void initState() {
    super.initState();
    _tabs = TabController(length: 2, vsync: this);
    _tabs.addListener(() => setState(() => _error = null));
  }

  @override
  void dispose() {
    _tabs.dispose();
    _loginUserCtrl.dispose();
    _loginPassCtrl.dispose();
    _regUserCtrl.dispose();
    _regPassCtrl.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    final isLogin = _tabs.index == 0;
    final key = isLogin ? _loginKey : _regKey;
    if (!key.currentState!.validate()) return;
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      if (isLogin) {
        await _api.userLogin(
            username: _loginUserCtrl.text.trim(),
            password: _loginPassCtrl.text);
      } else {
        await _api.register(
            username: _regUserCtrl.text.trim(),
            password: _regPassCtrl.text);
      }
      if (mounted) Navigator.of(context).pop(true);
    } catch (e) {
      setState(() {
        _loading = false;
        _error = e.toString().replaceFirst('Exception: ', '');
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      title: const Text('Access Sabil Network',
          style: TextStyle(fontWeight: FontWeight.bold, letterSpacing: -0.5)),
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(24)),
      content: SizedBox(
        width: 340,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            TabBar(
              controller: _tabs,
              tabs: const [Tab(text: 'Login'), Tab(text: 'Register')],
              labelColor: const Color(0xFF0F766E),
              indicatorColor: const Color(0xFF0F766E),
              unselectedLabelColor: Colors.grey,
              labelStyle: const TextStyle(fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 20),
            SizedBox(
              height: 180,
              child: TabBarView(
                controller: _tabs,
                children: [
                  Form(
                    key: _loginKey,
                    child: Column(
                      children: [
                        TextFormField(
                          controller: _loginUserCtrl,
                          decoration: const InputDecoration(
                              labelText: 'Username',
                              border: OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(12)))),
                          validator: (v) =>
                              v == null || v.trim().isEmpty ? 'Required' : null,
                          enabled: !_loading,
                        ),
                        const SizedBox(height: 10),
                        TextFormField(
                          controller: _loginPassCtrl,
                          decoration: const InputDecoration(
                              labelText: 'Password',
                              border: OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(12)))),
                          obscureText: true,
                          validator: (v) =>
                              v == null || v.isEmpty ? 'Required' : null,
                          enabled: !_loading,
                        ),
                      ],
                    ),
                  ),
                  Form(
                    key: _regKey,
                    child: Column(
                      children: [
                        TextFormField(
                          controller: _regUserCtrl,
                          decoration: const InputDecoration(
                              labelText: 'Username',
                              border: OutlineInputBorder(borderRadius: BorderRadius.all(Radius.circular(12)))),
                          validator: (v) {
                            if (v == null || v.trim().isEmpty) return 'Required';
                            if (v.trim().length < 3) return 'Min 3 chars';
                            if (!RegExp(r"^[a-zA-Z0-9_]+$")
                                .hasMatch(v.trim())) {
                              return 'Alpha-numerics only';
                            }
                            return null;
                          },
                          enabled: !_loading,
                        ),
                        const SizedBox(height: 10),
                        TextFormField(
                          controller: _regPassCtrl,

_StalenessInfo _stalenessInfo(String lastVerifiedAt) {
  final verified = DateTime.tryParse(lastVerifiedAt);
  if (verified == null) return const _StalenessInfo('Unknown status freshness', Colors.grey);
  final days = DateTime.now().difference(verified).inDays;
  if (days <= 3) {
    return _StalenessInfo(
        days == 0 ? 'Verified today' : 'Verified $days day${days == 1 ? '' : 's'} ago',
        const Color(0xFF10B981));
  }
  if (days <= 14) return _StalenessInfo('Stale: $days days ago', const Color(0xFFF59E0B));
  if (days <= 30) return _StalenessInfo('Stale: $days days ago', const Color(0xFFEF4444));
  final weeks = (days / 7).floor();
  return _StalenessInfo(
      '$weeks weeks ago — requires fresh verification', const Color(0xFFEF4444));
}

MediaType _getMediaType(String path) {
  final ext = path.split('.').last.toLowerCase();
  if (ext == 'png') return MediaType('image', 'png');
  if (ext == 'webp') return MediaType('image', 'webp');
  return MediaType('image', 'jpeg');
}

import 'dart:async';

import 'package:geolocator/geolocator.dart';

class LocationService {
  /// Resolves a fix without blocking indefinitely (emulators often stall on
  /// high-accuracy GPS). Uses last-known when available, then a bounded wait.
  Future<Position?> resolveCurrentPosition() async {
    final serviceEnabled = await Geolocator.isLocationServiceEnabled();
    if (!serviceEnabled) {
      throw LocationServiceDisabledException();
    }

    LocationPermission permission = await Geolocator.checkPermission();
    if (permission == LocationPermission.denied) {
      permission = await Geolocator.requestPermission();
      if (permission == LocationPermission.denied) {
        throw LocationPermissionDeniedException();
      }
    }

    if (permission == LocationPermission.deniedForever) {
      throw LocationPermissionPermanentlyDeniedException();
    }

    final cached = await Geolocator.getLastKnownPosition();
    if (cached != null) {
      return cached;
    }

    try {
      return await Geolocator.getCurrentPosition(
        desiredAccuracy: LocationAccuracy.medium,
        timeLimit: const Duration(seconds: 12),
      );
    } on TimeoutException {
      return null;
    }
  }
}

class LocationServiceDisabledException implements Exception {
  final String message = 'Location services are disabled.';
}

class LocationPermissionDeniedException implements Exception {
  final String message = 'Location permission denied. Showing default area.';
}

class LocationPermissionPermanentlyDeniedException implements Exception {
  final String message = 'Location permission permanently denied.';
}
