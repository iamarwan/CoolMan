import os
from datetime import datetime

# 1. Grab the exact date and time right now
current_time = datetime.now()
timestamp_for_name = current_time.strftime("%Y%m%d_%H%M%S")
readable_time = current_time.strftime("%Y-%m-%d %I:%M:%S %p")

PROJECT_ROOT = "."
# 2. Make the filename dynamic so it NEVER overwrites your older audits
OUTPUT_FILE = f"coolman_dump_{timestamp_for_name}.txt"

# 3. Added 'android' and '.gradle' to prevent massive token waste
IGNORE_DIRS = {
    '.git', '__pycache__', 'venv', '.dart_tool', 'build', 
    '.idea', '.vscode', 'ios', 'web', 'windows', 'macos', 'linux', 'android', '.gradle'
}

ALLOWED_EXTS = {
    '.dart', '.py', '.yaml', '.xml', '.json', '.md'
}

def generate_code_dump():
    with open(OUTPUT_FILE, 'w', encoding='utf-8') as outfile:
        # Stamp the top of the file with the readable date and time
        outfile.write("### SABIL / COOLMAN FULL CODEBASE DUMP ###\n")
        outfile.write(f"### GENERATED ON: {readable_time} ###\n\n")
        
        for dirpath, dirnames, filenames in os.walk(PROJECT_ROOT):
            dirnames[:] = [d for d in dirnames if d not in IGNORE_DIRS]
            
            for filename in filenames:
                ext = os.path.splitext(filename)[1].lower()
                
                if ext in ALLOWED_EXTS:
                    filepath = os.path.join(dirpath, filename)
                    relative_path = os.path.relpath(filepath, PROJECT_ROOT)
                    
                    # Skip the script itself and ANY previous dump files
                    if filename == "packer.py" or filename.startswith("coolman_dump_"):
                        continue
                    
                    # 4. Use XML tags so Claude actually understands where files begin and end
                    outfile.write(f'<document path="{relative_path}">\n')
                    
                    try:
                        with open(filepath, 'r', encoding='utf-8') as infile:
                            code_content = infile.read()
                            outfile.write(code_content)
                            # Ensure clean formatting
                            if not code_content.endswith('\n'):
                                outfile.write('\n')
                    except Exception as e:
                        outfile.write(f"// Error reading file: {e}\n")
                        
                    outfile.write(f'</document>\n\n')
                    
    print(f"✅ Audit packed and time-stamped: {OUTPUT_FILE}")

if __name__ == "__main__":
    generate_code_dump()
